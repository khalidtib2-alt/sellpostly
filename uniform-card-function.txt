export function StorefrontProducts({shop,products,slug,snapshot,onAdd,onPay,onCod,onShare,management}:Props){
 const {t}=useLanguage(),affiliate=shop.shopMode==='affiliate',[quick,setQuick]=useState<Product|null>(null),[demo,setDemo]=useState('');
 const title=(p:Product)=>p.id.startsWith('sample-')&&p.name in translations?t(p.name as Message):p.name;
 return <><div className={affiliate?'editorial-products':'boutique-products'}>{products.map((p,index)=>{
  const identity=extractProductIdentity(p.identityOverride||p.name,p.category),amazon=affiliate&&isAmazonDestination(p.url),restricted=affiliate&&amazonAffiliateProduct(p),discount=restricted?null:dealDiscount(p),soldOut=p.stock===0,highlights=productHighlights(p.description),stockPhoto=stockMetadata(p.stockPhoto,p.image);
  return <article id={'product-'+p.id} key={p.id} className={affiliate?'editorial-product'+(stockPhoto?' has-representative-photo':''):'boutique-product'}>
   <div className="store-card-visual aspect-[4/3] w-full"><span className="store-card-number">{String(index+1).padStart(2,'0')} / {t(affiliate?'THE EDIT':'COLLECTION')}</span>
    {affiliate?<div className="store-photo-frame h-full w-full">{p.image&&<img className="h-full w-full object-contain p-4" src={p.image} alt={title(p)} loading="lazy"/>}</div>:<a href={productPath(p.id,slug,snapshot)} className="store-photo-frame h-full w-full" aria-label={title(p)}>{p.image&&<img className="h-full w-full object-contain p-4" src={p.image} alt={title(p)} loading="lazy"/>}</a>}
    {affiliate&&stockPhoto&&<span className="studio-cover-caption">{t('Representative style photo')}</span>}{discount&&<span className="store-deal-badge">-{discount}%</span>}{!restricted&&p.badge&&p.badge!=='none'&&<span className="store-feature-badge">{t(p.badge==='hot'?'Hot':p.badge==='sale'?'Sale':p.badge==='limited'?'Limited':'New')}</span>}{!affiliate&&<button className="store-quick-view" type="button" onClick={()=>setQuick(p)}><Eye size={16}/>{t('Quick View')}</button>}
   </div>
   <div className="store-card-body">
    <div className="store-photo-credit">{affiliate&&stockPhoto&&<a href={stockPhoto.pageUrl} target="_blank" rel="noopener noreferrer">{t('Photo credit')}: {stockPhoto.photographer==='Pexels contributor'?t('Pexels contributor'):stockPhoto.photographer} / {stockPhoto.provider}</a>}</div>
    <p className="store-eyebrow">{categories.includes(p.category as typeof categories[number])?t(p.category as Message):p.category}{affiliate&&p.source?' · '+p.source:''}</p>
    <h3>{affiliate?(p.id.startsWith('sample-')?title(p):identity.confidence==='matched'?identity.coreName:title(p)):<a href={productPath(p.id,slug,snapshot)}>{title(p)}</a>}</h3>
    <div className="store-price-slot">{!restricted&&<p className="store-card-price">{p.price}{p.compareAtPrice&&<del>{p.compareAtPrice}</del>}</p>}</div>
    {affiliate?<ul className="store-highlights">{highlights.map((line,i)=><li key={i}><Check size={15}/><span>{line}</span></li>)}</ul>:<><p className="store-card-description">{p.description}</p><p className="store-stock">{p.stock!==undefined?(soldOut?t('Sold Out'):p.stock+' '+t('available')):'\u00a0'}</p></>}
    <div className="store-card-actions">
     {affiliate?<div className="store-primary-action">{safeUrl(p.url)&&<a className="store-buy-button" href={safeUrl(p.url)} target="_blank" rel="nofollow sponsored noopener noreferrer">{t(amazon?'Check Price on Amazon':'View Best Deal')}<ArrowUpRight size={18}/></a>}</div>:<><div className="store-personal-actions"><button type="button" disabled={soldOut} className="store-buy-button" onClick={()=>onAdd(p,1)}><ShoppingBag size={16}/>{t('Add to Cart')}</button><a className="store-detail-link" href={productPath(p.id,slug,snapshot)}>{t('View Product')}<ArrowRight size={16}/></a></div><div className="store-payment-links"><button type="button" disabled={soldOut} onClick={()=>onPay(p,1)}>{t('Payment Options')}</button>{shop.escrowEnabled!==false&&<button type="button" disabled={soldOut} onClick={()=>onPay(p,1)}>{t('Pay with USDC')}</button>}{shop.codEnabled!==false&&<button type="button" disabled={soldOut} onClick={()=>onCod(p,1)}>{t('Cash on Delivery (COD)')}</button>}</div></>}
     <button type="button" className="store-share-action text-sm font-semibold text-emerald-800 underline underline-offset-4" onClick={()=>onShare?.(p)}>{t('Copy / Share Link')}</button>
     <div className="store-video-slot">{promoVideo(p.promo_video_url)&&<button type="button" className="rounded-full border border-emerald-600 px-3 py-2 text-sm font-bold text-emerald-800" onClick={()=>setDemo(p.promo_video_url!)}>{t('Watch Demo')}</button>}</div>
     {management&&<div className="mt-4 flex flex-wrap gap-3 border-t pt-3 text-xs">{management(p)}</div>}
    </div>
   </div>
  </article>;
 })}</div>{demo&&<PromoVideoModal url={demo} onClose={()=>setDemo('')}/>} {quick&&!affiliate&&<QuickView shop={shop} product={quick} slug={slug} snapshot={snapshot} onClose={()=>setQuick(null)} onAdd={onAdd} onPay={onPay} onCod={onCod}/>}</>;
}
